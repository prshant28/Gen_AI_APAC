import { useEffect, useState } from "react";
import { Link, useLocation, useSearch } from "wouter";
import { motion, AnimatePresence } from "framer-motion";
import { ArrowRight, Eye, EyeOff, Mail, Lock, User as UserIcon, ShieldCheck, Sparkles, Loader2, ArrowLeft } from "lucide-react";
import { loginUser, registerUser, isUserLoggedIn, setUserToken, firebaseGoogleLogin } from "@/lib/api";
import { useToast } from "@/hooks/use-toast";
import BorderGlow from "@/components/BorderGlow";

export default function Login() {
  const [, setLocation] = useLocation();
  const search = useSearch();
  const { toast } = useToast();
  const [mode, setMode] = useState<"sign-in" | "sign-up">(
    () => new URLSearchParams(search).get("mode") === "signup" ? "sign-up" : "sign-in"
  );
  const [showPassword, setShowPassword] = useState(false);
  const [submitting, setSubmitting] = useState(false);
  const [form, setForm] = useState({
    fullName: "",
    email: "",
    phone: "",
    password: "",
  });

  // Handle Google OAuth redirect: ?google_token=xxx or ?error=google_failed
  useEffect(() => {
    const params = new URLSearchParams(search);
    const googleToken = params.get("google_token");
    const errorParam = params.get("error");

    if (googleToken) {
      setUserToken(googleToken);
      setLocation("/account");
      return;
    }
    if (errorParam === "google_failed") {
      toast({
        title: "Google sign-in failed",
        description: "Could not sign in with Google. Please try again.",
        variant: "destructive",
      });
      // Clean URL
      window.history.replaceState({}, "", window.location.pathname);
    }
  }, [search, setLocation, toast]);

  // Already signed in? Bounce to /account.
  useEffect(() => {
    if (isUserLoggedIn()) {
      setLocation("/account");
    }
  }, [setLocation]);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (submitting) return;
    setSubmitting(true);
    try {
      if (mode === "sign-in") {
        const res = await loginUser(form.email.trim(), form.password);
        if (!res.success) {
          toast({
            title: "Sign in failed",
            description: res.error || "Invalid email or password.",
            variant: "destructive",
          });
          return;
        }
        toast({ title: "Welcome back!", description: "Signed in successfully." });
        setLocation("/account");
      } else {
        if (!form.fullName.trim() || !form.email.trim() || !form.password) {
          toast({
            title: "Missing details",
            description: "Please fill name, email, and password.",
            variant: "destructive",
          });
          return;
        }
        const res = await registerUser({
          fullName: form.fullName.trim(),
          email: form.email.trim(),
          phone: form.phone.trim() || undefined,
          password: form.password,
        });
        if (!res.success) {
          toast({
            title: "Sign up failed",
            description: res.error || "Could not create your account.",
            variant: "destructive",
          });
          return;
        }
        toast({ title: "Account created", description: "Welcome to X247 Rewards." });
        setLocation("/account");
      }
    } finally {
      setSubmitting(false);
    }
  };

  const handleGoogle = async () => {
    if (submitting) return;
    setSubmitting(true);
    try {
      const res = await firebaseGoogleLogin();
      if (!res.success) {
        toast({
          title: "Google sign-in failed",
          description: res.error || "Could not sign in with Google.",
          variant: "destructive",
        });
        return;
      }
      setLocation("/account");
    } catch {
      toast({
        title: "Google sign-in failed",
        description: "Popup was closed or an error occurred.",
        variant: "destructive",
      });
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <div className="login-page">
      {/* Decorative background layer */}
      <div className="login-bg-glow" aria-hidden="true" />
      <div className="login-bg-grain" aria-hidden="true" />

      {/* Full-screen split card */}
      <div className="login-outer-card">
        {/* LEFT — Form panel */}
        <div className="login-form-panel">
          <div className="login-form-inner">
            {/* Back to home */}
            <Link href="/" className="login-home-btn" aria-label="Go to home">
              <ArrowLeft className="w-4 h-4" />
              <span>Home</span>
            </Link>

            {/* 3D glass card */}
            <div className="login-form-card">
            <div className="login-form-eyebrow">
              <span className="login-form-eyebrow-dot" />
              <span>ACCOUNT</span>
            </div>

            <AnimatePresence mode="wait">
              <motion.div
                key={`heading-${mode}`}
                initial={{ opacity: 0, y: 8 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, y: -6 }}
                transition={{ duration: 0.32, ease: [0.22, 1, 0.36, 1] }}
              >
                <h1 className="login-heading">
                  {mode === "sign-in" ? "Sign in to your account" : "Create your account"}
                </h1>
                <p className="login-sub">
                  {mode === "sign-in"
                    ? "Enter your credentials to view your rewards dashboard."
                    : "Join thousands of winners. Free forever, no credit card required."}
                </p>
              </motion.div>
            </AnimatePresence>

            {/* Google button */}
            <BorderGlow
              as="button"
              type="button"
              onClick={handleGoogle}
              borderRadius={14}
              glowRadius={18}
              cardBg="rgba(10,10,10,0.6)"
              className="premium-btn premium-btn-md premium-btn-ghost glass-btn-effect w-full login-google-btn-wrap"
              data-testid="button-login-google"
            >
              <svg className="login-google-icon relative z-[2]" viewBox="0 0 24 24" aria-hidden="true">
                <path fill="#EA4335" d="M12 5.04c1.95 0 3.7.67 5.07 1.98l3.78-3.78C18.42 1 15.46 0 12 0 7.31 0 3.26 2.69 1.28 6.61l4.42 3.42C6.74 7.04 9.14 5.04 12 5.04z" />
                <path fill="#4285F4" d="M23.49 12.27c0-.78-.07-1.53-.2-2.27H12v4.51h6.47c-.28 1.5-1.13 2.78-2.41 3.64l3.69 2.86c2.16-2 3.74-4.95 3.74-8.74z" />
                <path fill="#FBBC05" d="M5.7 14.2a7.06 7.06 0 0 1 0-4.4L1.28 6.4A12 12 0 0 0 0 12c0 1.94.46 3.78 1.28 5.4l4.42-3.2z" />
                <path fill="#34A853" d="M12 24c3.24 0 5.96-1.07 7.95-2.91l-3.69-2.86c-1.02.69-2.34 1.1-4.26 1.1-2.86 0-5.27-2-6.13-4.7l-4.42 3.42C3.26 21.31 7.31 24 12 24z" />
              </svg>
              <span className="relative z-[2]">Continue with Google</span>
            </BorderGlow>

            <div className="login-divider">
              <span className="login-divider-line" />
              <span className="login-divider-text">or</span>
              <span className="login-divider-line" />
            </div>

            {/* Email/password form */}
            <form onSubmit={handleSubmit} className="login-form" autoComplete="on">
              <AnimatePresence mode="wait">
                {mode === "sign-up" && (
                  <motion.div
                    key="signup-fields"
                    initial={{ opacity: 0, height: 0 }}
                    animate={{ opacity: 1, height: "auto" }}
                    exit={{ opacity: 0, height: 0 }}
                    transition={{ duration: 0.3 }}
                    style={{ overflow: "hidden" }}
                  >
                    <div className="login-field">
                      <label htmlFor="fullName" className="login-label">Full name</label>
                      <div className="login-input-wrap">
                        <UserIcon className="login-input-icon" />
                        <input
                          id="fullName"
                          name="fullName"
                          type="text"
                          autoComplete="name"
                          placeholder="Enter your full name"
                          value={form.fullName}
                          onChange={(e) => setForm({ ...form, fullName: e.target.value })}
                          className="login-input"
                          data-testid="input-fullname"
                        />
                      </div>
                    </div>
                  </motion.div>
                )}
              </AnimatePresence>

              <div className="login-field">
                <label htmlFor="email" className="login-label">Email address</label>
                <div className="login-input-wrap">
                  <Mail className="login-input-icon" />
                  <input
                    id="email"
                    name="email"
                    type="email"
                    autoComplete="email"
                    required
                    placeholder="Enter your email address"
                    value={form.email}
                    onChange={(e) => setForm({ ...form, email: e.target.value })}
                    className="login-input"
                    data-testid="input-email"
                  />
                </div>
              </div>

              <div className="login-field">
                <label htmlFor="password" className="login-label">Password</label>
                <div className="login-input-wrap">
                  <Lock className="login-input-icon" />
                  <input
                    id="password"
                    name="password"
                    type={showPassword ? "text" : "password"}
                    autoComplete={mode === "sign-in" ? "current-password" : "new-password"}
                    required
                    placeholder="Enter your password"
                    value={form.password}
                    onChange={(e) => setForm({ ...form, password: e.target.value })}
                    className="login-input"
                    data-testid="input-password"
                  />
                  <button
                    type="button"
                    onClick={() => setShowPassword((v) => !v)}
                    className="login-input-toggle"
                    aria-label={showPassword ? "Hide password" : "Show password"}
                  >
                    {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
                  </button>
                </div>
              </div>

              <BorderGlow
                as="button"
                type="submit"
                disabled={submitting}
                borderRadius={14}
                glowRadius={20}
                cardBg="rgba(6,6,6,0.95)"
                className={`premium-btn premium-btn-md glass-btn-effect w-full login-submit-btn-wrap${submitting ? " opacity-50 cursor-not-allowed" : ""}`}
                data-testid="button-submit"
              >
                {submitting ? (
                  <Loader2 className="w-4 h-4 animate-spin relative z-[2]" />
                ) : (
                  <>
                    <span className="relative z-[2]">{mode === "sign-in" ? "Sign in" : "Create account"}</span>
                    <ArrowRight className="w-4 h-4 relative z-[2] group-hover:translate-x-1 transition-transform" />
                  </>
                )}
              </BorderGlow>
            </form>

            <div className="login-mode-toggle">
              {mode === "sign-in" ? (
                <>
                  Don&apos;t have an account?{" "}
                  <button type="button" onClick={() => setMode("sign-up")} data-testid="link-toggle-signup">
                    Create one
                  </button>
                </>
              ) : (
                <>
                  Already have an account?{" "}
                  <button type="button" onClick={() => setMode("sign-in")} data-testid="link-toggle-signin">
                    Sign in
                  </button>
                </>
              )}
            </div>

            <div className="login-trust-row">
              <div className="login-trust-item">
                <ShieldCheck className="w-3 h-3" />
                <span>Bank-grade security</span>
              </div>
              <span className="login-trust-dot" />
              <div className="login-trust-item">
                <Sparkles className="w-3 h-3" />
                <span>Instant access</span>
              </div>
            </div>
            </div>{/* end login-form-card */}
          </div>
        </div>

        {/* RIGHT — Hero image panel */}
        <div className="login-img-panel">
          <img
            src={`${import.meta.env.BASE_URL}login-hero.png`}
            alt=""
            className="login-img-hero"
            aria-hidden="true"
          />
          <div className="login-img-overlay" aria-hidden="true" />
          <div className="login-img-watermark">x247</div>
        </div>
      </div>
    </div>
  );
}
